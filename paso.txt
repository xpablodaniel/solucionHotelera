1. Nueva función

En:

nano src/business/rooming.js

agregamos sin eliminar lo que ya funciona:

/**
 * Agrupa las reservas por habitación física.
 *
 * Una habitación puede contener pasajeros
 * pertenecientes a distintos vouchers.
 *
 * IMPORTANTE:
 * - No fusiona las reservas.
 * - Conserva el voucher de cada reserva.
 * - La habitación física es la unidad de Rooming.
 */
function groupRoomsForRooming(reservations) {

    if (!Array.isArray(reservations)) {
        throw new TypeError(
            "groupRoomsForRooming espera un array de reservas."
        );
    }


    const rooms = new Map();


    for (const reservation of reservations) {

        if (!reservation || typeof reservation !== "object") {
            continue;
        }


        const voucher =
            reservation.voucher ?? null;


        const habitaciones =
            Array.isArray(reservation.habitaciones)
                ? reservation.habitaciones
                : [];


        for (const room of habitaciones) {

            if (!room || !room.numero) {
                continue;
            }


            const numero = String(room.numero);


            if (!rooms.has(numero)) {

                rooms.set(numero, {
                    numero,
                    reservas: [],
                    pasajeros: [],
                    asignaciones: []
                });
            }


            const roomingRoom =
                rooms.get(numero);


            /*
             * Conservamos la reserva como unidad
             * independiente.
             */
            roomingRoom.reservas.push({

                voucher,

                pasajeros:
                    Array.isArray(room.pasajeros)
                        ? room.pasajeros
                        : []
            });


            /*
             * Los pasajeros pasan a formar parte
             * de la habitación física.
             */
            if (Array.isArray(room.pasajeros)) {

                roomingRoom.pasajeros.push(
                    ...room.pasajeros
                );
            }


            /*
             * Conservamos A/B/C sin convertirlas
             * en habitaciones diferentes.
             */
            if (Array.isArray(room.asignaciones)) {

                roomingRoom.asignaciones.push(
                    ...room.asignaciones
                );
            }
        }
    }


    return Array.from(rooms.values());
}
